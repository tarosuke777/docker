# ルートCA（最初だけ）
## 秘密鍵を作成
openssl genrsa -out rootCA.key 2048

## ルートCA証明書を作成（有効期限10年）
openssl req -x509 -new -nodes -key rootCA.key -sha256 -days 3650 \
  -out rootCA.crt \
  -subj "/CN=MyPrivateRootCA"

# 証明書の作成（各サイト毎）
## adg
openssl req -new -nodes -newkey rsa:2048 \
  -keyout adg.key \
  -out adg.csr \
  -subj "/CN=adg.home.arpa"

openssl x509 -req -in adg.csr \
  -CA rootCA.crt -CAkey rootCA.key -CAcreateserial \
  -out adg.crt -days 3650 -sha256 \
  -extfile <(printf "subjectAltName=DNS:adg.home.arpa")
