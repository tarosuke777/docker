# 証明書の作成
openssl req -x509 -nodes -days 3650 -newkey rsa:2048 \
  -keyout ./server.key \
  -out ./server.crt \
  -subj "/CN=www.home.arpa" \
  -addext "subjectAltName=DNS:www.home.arpa"
